    def _declared_mode_names(self) -> set[str]:
        """Introspect declared params to find every namespace under 'modes.*'."""
        names = set()
        for p in self._parameters.values():  # rclpy.node.Node._parameters dict
            if p.name.startswith('modes.') and p.name.count('.') >= 2:

                names.add(p.name.split('.')[1])
        return names

    def _resolve_active_mode(self) -> ModeView:
        name = self.get_parameter('active_mode').value
        # D-04 back-compat: if the bands are NaN (sentinel = "not declared in YAML"),
        # synthesize from legacy scalar params.
        bl = self.get_parameter(f'modes.{name}.band_low').value

                name=name,
                target=tgt,
                band_low=tgt - tol,
                band_high=tgt + tol,

    Step 1 — Replace `_ramp_setpoint(dt)` (lines 268–280) with `_ramp_setpoint_to_band(dt, mode)`:
    ```python
    def _ramp_setpoint_to_band(self, dt, mode: ModeView):
        """D-10: ramp toward the DEFENDED band edge, not midpoint."""
        ramp_seconds = self.get_parameter('pid_setpoint_ramp_seconds').value
        # Defended-edge target: low edge if defend_side in {low, both} and we're below;
        # high edge if defend_side in {high, both} and we're above; otherwise the nearer edge.
        rh = self.current_humidity
        if rh < mode.band_low:
            edge_target = mode.band_low
        elif rh > mode.band_high and mode.defend_side in ('high', 'both'):
            edge_target = mode.band_high
        else:
            # In-band or above-band-with-low-defense: park ramp at the nearest defended edge.
            # For defend_side=low this is band_low (we're trying to drift back into the band on a fall).
            edge_target = mode.band_low if mode.defend_side != 'high' else mode.band_high
        if ramp_seconds <= 0:
            self._effective_setpoint = edge_target
            return
        delta = edge_target - self._effective_setpoint
        if abs(delta) < 1e-6:
            self._effective_setpoint = edge_target
            return
        max_step = abs(delta) * (dt / ramp_seconds)
        step = max(-max_step, min(max_step, delta))
        self._effective_setpoint += step
    ```

    elif rh > mode.band_high:
