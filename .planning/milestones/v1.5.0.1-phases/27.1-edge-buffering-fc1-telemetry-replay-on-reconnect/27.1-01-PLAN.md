       ```javascript
       async function findTopicTimeDuplicates(pool, limit = 5) {
           const r = await pool.query(
               'SELECT topic, time, COUNT(*) AS n FROM telemetry ' +
               'GROUP BY 1,2 HAVING COUNT(*) > 1 ORDER BY n DESC LIMIT $1',

           );
           return r.rows;
       }

       async function applyTelemetryUniqueConstraint(pool) {
           // Idempotent — safe to run on every bridge boot.
           // Hypertable rule: UNIQUE/PK on a hypertable must include the partitioning column (time).
           // The (topic, time) tuple satisfies this. See RESEARCH §Pitfall 1.
           await pool.query(`
               DO $$
               BEGIN
                   IF NOT EXISTS (
                       SELECT 1 FROM pg_constraint
                       WHERE conname = 'telemetry_topic_time_unique'
                   ) THEN
                       ALTER TABLE telemetry
                           ADD CONSTRAINT telemetry_topic_time_unique UNIQUE (topic, time);
                   END IF;
               END $$;
           `);
       }

       module.exports = module.exports || {};
       module.exports.findTopicTimeDuplicates = findTopicTimeDuplicates;
       module.exports.applyTelemetryUniqueConstraint = applyTelemetryUniqueConstraint;
       ```
