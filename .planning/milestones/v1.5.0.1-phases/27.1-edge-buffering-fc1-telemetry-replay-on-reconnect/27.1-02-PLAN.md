<objective>
Add the `fc_buffer` ROS node to fc_core. It subscribes to every topic listed in `buffered_topics.yaml`, writes each message to a local SQLite WAL database, evicts rows older than 24h on a 60s timer, and exposes `GET /telemetry/since?ts=<ns>` over HTTP bound to the Tailscale interface for the bridge to poll on reconnect.

Purpose: This is the on-Pi half of the edge buffer. It captures every fc1 telemetry message regardless of whether the bridge is reachable, then serves the buffered points back on demand. Survives fc-core restarts (SQLite on disk). Stdlib only — no new pip deps.

Output: `fc_buffer` node runs alongside fc_sensors/fc_controller/fc_pwm_driver/fc_display/fc_camera in `fc.launch.py`, persists to `/var/lib/fc-core/buffer.sqlite`, serves HTTP on `100.96.239.75:8765` (Tailscale interface).
</objective>

       ```python
       #!/usr/bin/env python3
       """fc_buffer — local telemetry ring buffer + HTTP replay endpoint (Phase 999.1)."""
       import json, sqlite3, threading, http.server, socketserver, re
       from urllib.parse import urlparse, parse_qs
       from pathlib import Path
       import yaml

       CAMERA_PATTERN = re.compile(r'/fc1/camera')
       HTTP_LIMIT_CAP = 50000

       def init_schema(db):
           db.execute('PRAGMA journal_mode=WAL')
           db.execute('PRAGMA synchronous=NORMAL')
           db.execute('''CREATE TABLE IF NOT EXISTS telemetry_buffer (
               topic TEXT NOT NULL, time_ns INTEGER NOT NULL,
               value REAL, extra TEXT,
               PRIMARY KEY (topic, time_ns)
           )''')
           db.commit()

       def _write_row(db, topic, time_ns, value, extra):
           db.execute(
               'INSERT OR IGNORE INTO telemetry_buffer (topic, time_ns, value, extra) VALUES (?,?,?,?)',
               (topic, int(time_ns), float(value) if value is not None else None, extra)
           )
           db.commit()

       def _prune(db, retention_seconds, now_ns):
           cutoff_ns = int(now_ns) - int(retention_seconds) * 1_000_000_000
           db.execute('DELETE FROM telemetry_buffer WHERE time_ns < ?', (cutoff_ns,))
           db.commit()

       def _serve_since(db_path, since_ns, limit=10000):
           since_ns = int(since_ns)  # raises ValueError on bad input — caller handles
           limit = min(int(limit), HTTP_LIMIT_CAP)
           ro = sqlite3.connect(f'file:{db_path}?mode=ro', uri=True)
           try:
               rows = list(ro.execute(
                   'SELECT topic, time_ns, value, extra FROM telemetry_buffer '
                   'WHERE time_ns > ? ORDER BY time_ns LIMIT ?', (since_ns, limit)
               ))
           finally:
               ro.close()
           return [{'topic': r[0], 'time_ns': r[1], 'value': r[2], 'extra': r[3]} for r in rows]

       def _load_topics(yaml_path):
           data = yaml.safe_load(open(yaml_path))
           topics = data.get('buffered', [])
           for t in topics:
               if CAMERA_PATTERN.search(t['ros_topic']):
                   raise ValueError(f"camera topic forbidden in buffer: {t['ros_topic']}")
           return topics

       def _extract(bridge_topic, msg):
           """Return (time_ns, value_float, extra_json_or_none)."""
           # Header timestamp if present and non-zero, else fall back to now_ns from caller.
           # NB: caller passes now_ns via attached attribute; pure helper takes msg only.
           import time
           if hasattr(msg, 'header') and getattr(msg.header.stamp, 'sec', 0):
               ts_ns = msg.header.stamp.sec * 1_000_000_000 + msg.header.stamp.nanosec
           else:
               ts_ns = time.time_ns()
           # Type discrimination by attribute presence (avoids importing ROS message classes in tests):
           if hasattr(msg, 'relative_humidity'):
               extra = json.dumps({'frame_id': msg.header.frame_id}) if getattr(msg.header, 'frame_id', '') else None
               return ts_ns, msg.relative_humidity * 100.0, extra
           if hasattr(msg, 'temperature'):
               extra = json.dumps({'frame_id': msg.header.frame_id}) if getattr(msg.header, 'frame_id', '') else None
               return ts_ns, float(msg.temperature), extra
           if hasattr(msg, 'data'):
               if isinstance(msg.data, bool):
                   return ts_ns, 1.0 if msg.data else 0.0, None
               return ts_ns, float(msg.data), None
           raise TypeError(f'unhandled msg shape for {bridge_topic}')
       ```
