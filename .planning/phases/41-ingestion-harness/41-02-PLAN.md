```js
async function runFixtureThroughPipeline(fixture, { extractor, transcribe, logger }) {
  const imagePaths = (fixture.attachments || []).filter(a => a.type === 'image').map(a => a.path);
  const audioPaths = (fixture.attachments || []).filter(a => a.type === 'audio').map(a => a.path);

  let transcript = '';
  let transcribe_latency_ms = 0;
  if (audioPaths.length > 0) {
    const t0 = Date.now();
    const parts = [];
    for (const ap of audioPaths) {
      const r = await transcribe.transcribe({ audioPath: ap });
      parts.push(r.text || '');
    }
    transcript = parts.join('\n').trim();
    transcribe_latency_ms = Date.now() - t0;
  }

  const imageBlocks = await loadImageBlocks(imagePaths, logger);

  // Build the synthetic Signal-envelope shape the extractor expects.
  // fixture.envelope.body is the human text; harness appends transcript so the
  // extractor sees text+audio fused exactly as the live path does.
  const composedText = [fixture.envelope.body, transcript].filter(Boolean).join('\n').trim();

  const t1 = Date.now();
  const result = await extractor.extract({
    text: composedText,
    imageBlocks,
    fixtureName: fixture.name,
  });
  const extract_latency_ms = Date.now() - t1;

  return {
    fixture_id: fixture.name,
    kind: fixture.kind,
    session_id: fixture.session_id || null,
    expected: fixture.expected,
    actual: {
      ok: !!result.ok,
      draft: result.draft || null,
      per_field_confidence: result.per_field_confidence || {},
    },
    tokens: result.tokens || null,
    cost_usd: result.cost_usd || 0,
    transcribe_latency_ms,
    extract_latency_ms,
    harness_confirmed: true,
    error: result.ok ? null : (result.reason || 'unknown'),
  };
}

module.exports = { runFixtureThroughPipeline };
```

The adapter assumes `extractor.extract` accepts a `{text, imageBlocks, fixtureName}` payload. If the real extractor signature differs (Phase 38 may use a slightly different shape), the adapter is the SOLE place that translates -- live-mode wraps the real extractor in a tiny shim to expose this signature. (Plan 06 will exercise this against the real createExtractor; if a shim is needed it lives in run-harness.js wiring, not in the adapter.)
