<objective>
Refactor the bridge's `insertTelemetry()` to accept optional `tsMs` and `tsNs` parameters so backfilled rows can carry their original DDS timestamp; bundle the live-path improvement to use `msg.header.stamp` for RelativeHumidity/Temperature subscriptions; add a new `buffer_replay.js` module that polls fc1's `/telemetry/since` endpoint every 30s with a 15s HTTP timeout and INSERTs results with `ON CONFLICT (topic, time) DO NOTHING`; persist `last_ingested_ns` to a host-mounted volume so state survives `docker compose up -d --build bridge`.

Purpose: This is the elder-plops half of the edge buffer. After Plan 02 ships, fc1 holds buffered points; this plan teaches the bridge to fetch them on a steady cadence and write them to Timescale at their original timestamps so OpenMCT charts render gap-fill instead of holes. The 15s timeout (Pitfall 5) and the live-path timestamp fix (Pitfall 2) are load-bearing — without them the gap fills with wrong times or the loop hangs during the very outage we're solving.

CRITICAL CONTRACT: every one of the 9 live call sites of insertTelemetry MUST pass tsNs as the 4th argument. If any call site is missed, tests still pass (tsNs defaults to null and silently skips advanceLastIngested) but the system silently regresses RESEARCH §Q5 — first post-restart poll re-fetches the full 24h buffer. The grep acceptance criteria in Task 2 enforce this.

Output: bridge container has a polling module that fills gaps within ~60s of fc1 reachability returning; ~13min outages produce ~zero permanent data loss; backfilled points appear at original timestamps in Mission Control's history charts.

    1. Create `src/mission-control/bridge/src/buffer_replay.js`:

       ```javascript
       const http = require('http');
       const fs = require('fs');

       const DEFAULT_FC1_URL = process.env.FC1_BUFFER_URL || 'http://100.96.239.75:8765';
       const DEFAULT_STATE_FILE = process.env.BUFFER_REPLAY_STATE || '/var/lib/bridge/buffer-replay.state.json';
       const DEFAULT_INTERVAL_MS = 30 * 1000;
       const HTTP_TIMEOUT_MS = 15 * 1000;
       const BATCH_LIMIT = 10000;

       function loadLastTs(stateFile) {
           try { return JSON.parse(fs.readFileSync(stateFile, 'utf8')).last_ingested_ns || 0; }
           catch { return 0; }
       }

       function saveLastTs(stateFile, ns) {
           try {
               fs.mkdirSync(require('path').dirname(stateFile), { recursive: true });
               fs.writeFileSync(stateFile, JSON.stringify({ last_ingested_ns: ns, updated_at: Date.now() }));
           } catch (e) {
               console.error('[buffer-replay] saveLastTs failed:', e.message);
           }
       }

       function advanceLastIngested(stateFile, ns) {
           const current = loadLastTs(stateFile);
           if (ns > current) saveLastTs(stateFile, ns);
       }

       function parseNdjson(body) {
           if (!body) return [];
           return body.split('\n').filter(l => l.trim().length > 0).map(JSON.parse);
       }

       function defaultFetchFn(url) {
           return new Promise((resolve, reject) => {
               const req = http.get(url, (res) => {
                   if (res.statusCode !== 200) {
                       res.resume();
                       return reject(new Error(`HTTP ${res.statusCode}`));
                   }
                   let body = '';
                   res.setEncoding('utf8');
                   res.on('data', (chunk) => { body += chunk; });
                   res.on('end', () => resolve(body));
               });
               req.setTimeout(HTTP_TIMEOUT_MS, () => req.destroy(new Error('timeout')));
               req.on('error', reject);
           });
       }

       async function pollOnce({ pool, fc1Url = DEFAULT_FC1_URL, stateFile = DEFAULT_STATE_FILE, fetchFn = defaultFetchFn }) {
           const sinceNs = loadLastTs(stateFile);
           const url = `${fc1Url}/telemetry/since?ts=${sinceNs}&limit=${BATCH_LIMIT}`;
           const body = await fetchFn(url);
           const lines = parseNdjson(body);
           if (lines.length === 0) return { rows: 0, maxTs: sinceNs };
           let maxTs = sinceNs;
           const client = await pool.connect();
           try {
               await client.query('BEGIN');
               for (const r of lines) {
                   // r.extra is intentionally NOT propagated to Timescale — matches live-path
                   // behavior per RESEARCH §Q4 (live inserts only write time/topic/value).
                   const tsMs = Math.floor(r.time_ns / 1_000_000);
                   await client.query(
                       `INSERT INTO telemetry (time, topic, value)
                        VALUES (to_timestamp($1::double precision / 1000), $2, $3)
                        ON CONFLICT (topic, time) DO NOTHING`,
                       [tsMs, r.topic, r.value]
                   );
                   if (r.time_ns > maxTs) maxTs = r.time_ns;
               }
               await client.query('COMMIT');
           } catch (e) {
               await client.query('ROLLBACK');
               throw e;
           } finally {
               client.release();
           }
           saveLastTs(stateFile, maxTs);
           return { rows: lines.length, maxTs };
       }

       function start({ pool, fc1Url = DEFAULT_FC1_URL, stateFile = DEFAULT_STATE_FILE, intervalMs = DEFAULT_INTERVAL_MS }) {
           console.log(`[buffer-replay] polling ${fc1Url}/telemetry/since every ${intervalMs}ms, state=${stateFile}`);
           return setInterval(() => {
               pollOnce({ pool, fc1Url, stateFile }).then(r => {
                   if (r.rows > 0) console.log(`[buffer-replay] backfilled ${r.rows} rows up to ts_ns=${r.maxTs}`);
               }).catch(e => console.error('[buffer-replay] poll failed:', e.message));
           }, intervalMs);
       }

       module.exports = { loadLastTs, saveLastTs, advanceLastIngested, parseNdjson, pollOnce, start, DEFAULT_FC1_URL, DEFAULT_STATE_FILE };
       ```
