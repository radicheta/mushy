3. **Bucket interval function** (per D-06 — server-side downsampling):
```javascript
function bucketInterval(rangeMs) {
    const ONE_HOUR = 3600000;
    if (rangeMs <= 2 * ONE_HOUR)  return '1 minute';
    if (rangeMs <= 12 * ONE_HOUR) return '5 minutes';
    return '15 minutes';
}
```

Threshold rationale: <=2h returns ~120 points at 1min buckets, <=12h returns ~144 points at 5min, >12h returns ~96/day at 15min. All well under browser rendering limits.

4. **History endpoint** (per D-04):
```javascript
app.get('/history/:topic', async (req, res) => {
    const { topic } = req.params;

    // Validate topic against allowlist (prevents SQL injection via topic — T-07-01)
    if (!ALLOWED_TOPICS.includes(topic)) {
        return res.status(400).json({ error: 'Invalid topic' });
    }

    const start = parseInt(req.query.start, 10);
    const end   = parseInt(req.query.end,   10);
    if (!start || !end || isNaN(start) || isNaN(end)) {
        return res.status(400).json({ error: 'start and end query params required (ms epoch)' });
    }

    // Cap max range at 30 days to prevent unbounded queries (T-07-02 mitigation)
    const MAX_RANGE = 30 * 24 * 3600000;
    if (end - start > MAX_RANGE) {
        return res.status(400).json({ error: 'Max range is 30 days' });
    }

    if (!dbReady) {
        return res.status(503).json({ error: 'Database not available' });
    }

    const rangeMs  = end - start;
    const interval = bucketInterval(rangeMs);

    try {
        const result = await pool.query(
            `SELECT time_bucket($1::interval, time) AS bucket,
                    AVG(value) AS value
             FROM telemetry
             WHERE topic = $2
               AND time >= $3
               AND time <= $4
             GROUP BY bucket
             ORDER BY bucket ASC`,
            [interval, topic, new Date(start), new Date(end)]
        );
        const datums = result.rows.map(row => ({
            value: parseFloat(row.value),
            utc:   new Date(row.bucket).getTime()
        }));
        res.json(datums);
    } catch (err) {
        console.error('[db] history query failed:', err.message);
        res.status(500).json({ error: 'Query failed' });
    }
});
```

Key details:
- Topic is validated against `ALLOWED_TOPICS` before hitting SQL (never interpolated)
- `start` and `end` are parsed as integers (ms epoch) and passed as `new Date()` to pg (per Research Pitfall 5)
- Response format is `[{value: number, utc: number}]` — exactly what OpenMCT expects
- 30-day max range prevents accidental full-table scans
- `parseFloat(row.value)` ensures AVG result is a JS number, not a string

**Change 1: Add historyUrl option and wire request() (replacing the stub at line 214)**

At the top of the `install` function (around line 70), add `historyUrl` extraction:

var historyUrl = (options && options.historyUrl) || 'http://localhost:8081/history';

Replace the `request` function (line 214, currently `return Promise.resolve([])`) with:
```javascript
request: function (domainObject, options) {
    var sensor = SENSORS.find(function (s) {
        return s.identifier.key === domainObject.identifier.key;
    });
    if (!sensor) return Promise.resolve([]);

    var url = historyUrl
        + '/' + encodeURIComponent(sensor.identifier.key)
        + '?start=' + options.start
        + '&end='   + options.end;

    return fetch(url)
        .then(function (resp) {
            if (!resp.ok) return [];
            return resp.json();
        })
        .catch(function () { return []; });
},
```

This uses `sensor.identifier.key` (e.g., `fc.humidity`) as the URL path segment — this matches the `topic` column in the DB and the `ALLOWED_TOPICS` list in the bridge. `options.start` and `options.end` are ms epoch integers from OpenMCT's time conductor.

The `.catch(function () { return []; })` ensures that if the bridge is down or DB is unavailable, OpenMCT shows an empty chart rather than erroring.

**Change 2: Update WebSocket onmessage handler to work with raw broadcast format**

The current `onmessage` handler (lines 167-175) expects rosbridge protocol: `{op: 'publish', topic: '/fc1/humidity', msg: {...}}`. The new index.js broadcasts raw objects: `{humidity: 72.5, timestamp: 1743900000000}`.

Replace the `ws.onmessage` function (lines 167-175) with a dispatcher that maps field names to SENSORS:
```javascript
ws.onmessage = function (event) {
    var data;
    try { data = JSON.parse(event.data); } catch (e) { return; }

    // Map raw broadcast field names to ROS topics for dispatch
    var fieldToTopic = {
        humidity:    '/fc1/humidity',
        temperature: '/fc1/temperature',
        co2:         '/fc1/co2',
        humidifier:  '/fc1/actuators/humidifier'
    };

    var matchedTopic = null;
    var rawValue = null;
    Object.keys(fieldToTopic).forEach(function (field) {
        if (data[field] !== undefined) {
            matchedTopic = fieldToTopic[field];
            rawValue = data[field];
        }
    });

    if (matchedTopic && subs[matchedTopic]) {
        // Build a synthetic msg object that sensor.extract() can process
        var sensor = SENSORS.find(function (s) { return s.topic === matchedTopic; });
        if (!sensor) return;

        // Invoke each subscriber's handler with a datum (value + utc)
        // directly, bypassing extract() since we already have the value
        subs[matchedTopic].forEach(function (cb) {
            try {
                // The subscribe handler (line 223) calls sensor.extract(msg)
                // We need to pass a msg shape that extract() returns the right value.
                // For humidity: extract = msg.relative_humidity * 100, and bridge sends {humidity: value*100}
                // So we construct a msg that produces the same result through extract():
                var syntheticMsg = {};
                if (matchedTopic === '/fc1/humidity') {
                    syntheticMsg.relative_humidity = rawValue / 100;
                    syntheticMsg.header = { stamp: { sec: Math.floor(data.timestamp / 1000), nanosec: (data.timestamp % 1000) * 1e6 } };
                } else if (matchedTopic === '/fc1/temperature') {
                    syntheticMsg.temperature = rawValue;
                    syntheticMsg.header = { stamp: { sec: Math.floor(data.timestamp / 1000), nanosec: (data.timestamp % 1000) * 1e6 } };
                } else if (matchedTopic === '/fc1/co2') {
                    syntheticMsg.data = rawValue;
                } else if (matchedTopic === '/fc1/actuators/humidifier') {
                    syntheticMsg.data = rawValue === 1;
                }
                cb(syntheticMsg);
            } catch (e) { /* swallow per-handler errors */ }
        });
    }
};
```

This approach preserves the existing `subscribe` handler (line 217-234) which calls `sensor.extract(msg)` and `getTimestamp(msg)`. The synthetic msg objects are shaped so that each sensor's `extract()` function and `getTimestamp()` return the correct values.
