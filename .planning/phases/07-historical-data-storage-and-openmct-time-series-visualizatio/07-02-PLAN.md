function bucketInterval(rangeMs) {
    const ONE_HOUR = 3600000;
    if (rangeMs <= 2 * ONE_HOUR)  return '1 minute';
    if (rangeMs <= 12 * ONE_HOUR) return '5 minutes';
    return '15 minutes';
}

Threshold rationale: <=2h returns ~120 points at 1min buckets, <=12h returns ~144 points at 5min, >12h returns ~96/day at 15min. All well under browser rendering limits.

4. **History endpoint** (per D-04):
```javascript
app.get('/history/:topic', async (req, res) => {
    const { topic } = req.params;

    // Validate topic against allowlist (prevents SQL injection via topic — T-07-01)
    if (!ALLOWED_TOPICS.includes(topic)) {
        return res.status(400).json({ error: 'Invalid topic' });
    }

    const start = parseInt(req.query.start, 10);
    const end   = parseInt(req.query.end,   10);
    if (!start || !end || isNaN(start) || isNaN(end)) {
        return res.status(400).json({ error: 'start and end query params required (ms epoch)' });
    }

    // Cap max range at 30 days to prevent unbounded queries (T-07-02 mitigation)
    const MAX_RANGE = 30 * 24 * 3600000;
    if (end - start > MAX_RANGE) {
        return res.status(400).json({ error: 'Max range is 30 days' });
    }

    if (!dbReady) {
        return res.status(503).json({ error: 'Database not available' });
    }

    const rangeMs  = end - start;
    const interval = bucketInterval(rangeMs);

    try {
        const result = await pool.query(
            `SELECT time_bucket($1::interval, time) AS bucket,
                    AVG(value) AS value
             FROM telemetry
             WHERE topic = $2
               AND time >= $3
               AND time <= $4
             GROUP BY bucket
             ORDER BY bucket ASC`,
            [interval, topic, new Date(start), new Date(end)]
        );
        const datums = result.rows.map(row => ({
            value: parseFloat(row.value),
            utc:   new Date(row.bucket).getTime()
        }));
        res.json(datums);
    } catch (err) {
        console.error('[db] history query failed:', err.message);
        res.status(500).json({ error: 'Query failed' });
    }
});
```

Key details:
- Topic is validated against `ALLOWED_TOPICS` before hitting SQL (never interpolated)
- `start` and `end` are parsed as integers (ms epoch) and passed as `new Date()` to pg (per Research Pitfall 5)
- Response format is `[{value: number, utc: number}]` — exactly what OpenMCT expects
- 30-day max range prevents accidental full-table scans
- `parseFloat(row.value)` ensures AVG result is a JS number, not a string

request: function (domainObject, options) {
    var sensor = SENSORS.find(function (s) {
        return s.identifier.key === domainObject.identifier.key;
    });
    if (!sensor) return Promise.resolve([]);

    var url = historyUrl

**Change 2: Update WebSocket onmessage handler — emit datums directly (no synthetic msg reconstruction)**

The current `onmessage` handler (lines 167-175) expects rosbridge protocol: `{op: 'publish', topic: '/fc1/humidity', msg: {...}}`. The new index.js broadcasts raw objects: `{humidity: 72.5, timestamp: 1743900000000}`.

Replace the `ws.onmessage` function (lines 167-175) with a dispatcher that maps field names to datums directly. IMPORTANT: Do NOT reconstruct synthetic msg objects and pass them through sensor.extract() — that would invert the bridge's transform (e.g., bridge sends `humidity * 100`, then synthetic msg would divide by 100, then extract() would multiply by 100 again). Instead, emit `{value, utc}` datums directly since the broadcast values are already in display units:

```javascript
ws.onmessage = function (event) {
    var data;
    try { data = JSON.parse(event.data); } catch (e) { return; }

    // Map raw broadcast field names to sensor keys for dispatch
    var fieldToKey = {
        humidity:    'fc.humidity',
        temperature: 'fc.temperature',
        co2:         'fc.co2',
        humidifier:  'fc.humidifier'
    };

    Object.keys(fieldToKey).forEach(function (field) {
        if (data[field] !== undefined) {
            var key = fieldToKey[field];
            var sensor = SENSORS.find(function (s) { return s.identifier.key === key; });
            if (!sensor) return;

            // Build datum directly — broadcast values are already in display units
            // (bridge sends humidity*100, temperature in C, co2 as float, humidifier as 0/1)
            // This avoids double-transform: do NOT reconstruct a synthetic ROS msg
            // and pass through sensor.extract()
            var datum = {
                value: data[field],
                utc: data.timestamp
            };

            // The subscribe callback (line ~223) currently calls sensor.extract(msg)
            // and getTimestamp(msg). We must update that callback to also accept
            // pre-built datums. The simplest approach: check if the callback's
            // subscriber list uses the topic-based dispatch.
            if (subs[sensor.topic]) {
                subs[sensor.topic].forEach(function (cb) {
                    try { cb(datum); } catch (e) { /* swallow per-handler errors */ }
                });
            }
        }
    });
};
```

**Also update the subscribe handler** (around line 217-234) so it can accept both raw datums `{value, utc}` from the new onmessage AND still work if someone passes a ROS msg. The subscribe handler currently does:
```javascript
cb({ value: sensor.extract(msg), utc: getTimestamp(msg) });
```

Change it to check whether the argument is already a datum:
```javascript
// In the subscribe callback that wraps sensor.extract + getTimestamp:
function (msgOrDatum) {
    if (msgOrDatum.utc !== undefined && msgOrDatum.value !== undefined) {
        // Already a datum from the new onmessage handler
        cb(msgOrDatum);
    } else {
        // Legacy ROS msg — extract value and timestamp
        cb({ value: sensor.extract(msgOrDatum), utc: getTimestamp(msgOrDatum) });
    }
}
```

This approach:
- Avoids the double-transform problem (bridge multiplies humidity by 100, we use that value directly)
- Is forward-compatible if rosbridge messages ever come back
- Keeps the subscribe/request datum contract identical: `{value: number, utc: number}`
