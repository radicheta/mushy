    ```python
    def _validate_params(self, params: list[Parameter]) -> SetParametersResult:
        # Build "post-batch view" so cross-param invariants check the WOULD-BE state
        # after the batch is applied (Pitfall 4 atomicity: whole batch passes or fails).
        post = {p.name: p.value for p in params}
        def get_post(name, fallback_param=None):
            if name in post:
                return post[name]
            return self.get_parameter(fallback_param or name).value

        republish_current_mode = False
        for p in params:
            n = p.name
            v = p.value
            # Band invariants
            if n.startswith('modes.') and n.endswith('.band_low'):
                prefix = n.rsplit('.', 1)[0]
                bh = get_post(f'{prefix}.band_high')
                if not (0.0 <= v < bh <= 1.0):
                    return SetParametersResult(successful=False, reason=f'{n}: must satisfy 0<=band_low<band_high<=1 (got band_low={v}, band_high={bh})')
                republish_current_mode = True
            elif n.startswith('modes.') and n.endswith('.band_high'):
                prefix = n.rsplit('.', 1)[0]
                bl = get_post(f'{prefix}.band_low')
                if not (0.0 <= bl < v <= 1.0):
                    return SetParametersResult(successful=False, reason=f'{n}: must satisfy 0<=band_low<band_high<=1 (got band_low={bl}, band_high={v})')
                republish_current_mode = True
            elif n.startswith('modes.') and n.endswith('.defend_side'):
                if v not in ('low', 'high', 'both'):
                    return SetParametersResult(successful=False, reason=f'{n}: must be one of low|high|both (got {v!r})')
                republish_current_mode = True
            elif n.startswith('modes.') and n.endswith('.target_humidity'):
                if not (0.0 <= v <= 1.0):
                    return SetParametersResult(successful=False, reason=f'{n}: must be in [0,1] (got {v})')
                republish_current_mode = True
            elif n == 'active_mode':
                declared = self._declared_mode_names()
                if v not in declared:
                    return SetParametersResult(successful=False, reason=f'active_mode={v!r} not in declared modes {sorted(declared)}')
                republish_current_mode = True
            # Defense-in-depth PID range bounds (mirrors bridge allowlist; same bounds
            # locked here so a bridge bypass cannot push insane gains).
            elif n == 'pid_kp':
                if not (0.0 <= v <= 5.0):
                    return SetParametersResult(successful=False, reason=f'pid_kp must be in [0,5] (got {v})')
            elif n == 'pid_ki':
                if not (0.0 <= v <= 1.0):
                    return SetParametersResult(successful=False, reason=f'pid_ki must be in [0,1] (got {v})')
            elif n == 'pid_kd':
                if not (0.0 <= v <= 20.0):
                    return SetParametersResult(successful=False, reason=f'pid_kd must be in [0,20] (got {v})')

        # Schedule republish AFTER the batch lands. rclpy applies params after this
        # callback returns successful=True, so we use a one-shot timer to avoid
        # publishing pre-applied state.
        if republish_current_mode:
            # Use a flag the next tick observes. rclpy applies the new param values AFTER
            # this callback returns successful=True; publishing inside the callback would
            # emit the OLD ModeView. The control_loop drains this flag on the next tick.
            self._pending_current_mode_republish = ('param_set',)

        return SetParametersResult(successful=True)
    ```

    def _handle_set_mode(self, request, response):
