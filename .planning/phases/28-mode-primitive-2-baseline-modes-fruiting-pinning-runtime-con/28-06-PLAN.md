<!-- Endpoint contract — separate from /control/param per D-19 (explicit Save-to-repo). -->

`POST /control/persist`
  Body: same shape as /control/param: `{node, param, value}` or `{node, params: []}`
  Behavior: validates with the SAME ALLOWLIST as /control/param, then writes the (param, value) tuple(s) into `/var/lib/fc-core/runtime_overrides.yaml` on fc1. Does NOT call SetParameters — that's plan 05's job; persist is reboot-survival ONLY.
  Responses:
    200  `{ok: true, persisted: [{param, value}, ...], path: '/var/lib/fc-core/runtime_overrides.yaml'}`
    400  `{error: ...}` — same allowlist/validation rules as /control/param
    500  `{error: ...}` — transport failure (SSH or fc_buffer HTTP)

Overlay yaml shape on disk (RESEARCH §Open Question 4 recommendation):
```yaml
# AUTO-GENERATED by bridge POST /control/persist — do not hand-edit.
# Last write: <ISO 8601 timestamp>
fc_controller:
  ros__parameters:
    modes.fruiting.band_low: 0.94
    modes.fruiting.band_high: 0.98
    pid_kp: 0.40
```

`fc_config.yaml` is loaded FIRST by the launch (plan 07); `runtime_overrides.yaml` is loaded SECOND. ROS2 launch's `parameters=[file1, file2]` is last-wins for duplicate keys. `t_target: .nan` is preserved as YAML literal `.nan` per js-yaml encoding (verify: js-yaml serializes `NaN` to `.nan`; if it serializes to `nan` or `NaN`, plan 07's launch parse will fail — fall back to writing the literal string `.nan` via templated assembly rather than dump).

Transport branch (locked in 28-01-SPIKE.md §B):
- **If SSH-from-bridge confirmed:** bridge writes overlay locally to a temp file, scps to fc1's `/var/lib/fc-core/runtime_overrides.yaml.tmp`, then `ssh ubuntu@172.16.10.5 'mv runtime_overrides.yaml runtime_overrides.yaml.bak 2>/dev/null; mv runtime_overrides.yaml.tmp runtime_overrides.yaml'`.
- **If fc_buffer HTTP relay locked:** bridge POSTs the rendered yaml content to `http://172.16.10.5:8765/control/persist`; fc_buffer (which already runs on fc1 as user `ubuntu`) writes the file with the same atomic rename + .bak primitive locally.

    ```javascript
    // Phase 28 D-17 Layer 2: persistence (explicit Save-to-repo per D-19).
    // Same allowlist as control_param.js — Layer 1 and Layer 2 share validation rules.
    const yaml = require('js-yaml');
    const path = require('path');
    const cp = require('./control_param');

    const OVERLAY_PATH = process.env.FC_RUNTIME_OVERLAY || '/var/lib/fc-core/runtime_overrides.yaml';

    // mergeOverlay: takes existing parsed yaml object (may be {} or null) and an array of
    // {param, value} entries. Returns the new yaml object with fc_controller.ros__parameters
    // populated as flat dotted-key entries (D-03).
    function mergeOverlay(existing, entries) {
      const out = JSON.parse(JSON.stringify(existing || {}));  // deep clone, safe for plain data
      out.fc_controller = out.fc_controller || {};
      out.fc_controller.ros__parameters = out.fc_controller.ros__parameters || {};
      const params = out.fc_controller.ros__parameters;
      for (const { param, value } of entries) {
        // NaN handling: js-yaml dumps NaN as `.nan`; we just store the JS NaN here.
        params[param] = value;
      }
      return out;
    }

    function renderOverlay(obj) {
      const ts = new Date().toISOString();
      const header =
        '# AUTO-GENERATED by bridge POST /control/persist — do not hand-edit.\n' +
        `# Last write: ${ts}\n`;
      // js-yaml options: noCompatMode preserves `.nan` literal; lineWidth keeps long
      // values on one line; sortKeys=true makes byte-equality testable for idempotency.
      let body = yaml.dump(obj, { noCompatMode: false, lineWidth: 200, sortKeys: true });
      // Defensive: if js-yaml emitted `NaN` or `nan` instead of `.nan`, normalize.
      body = body.replace(/: NaN$/gm, ': .nan').replace(/: nan$/gm, ': .nan');
      return header + body;
    }

    function makeHandler(transport, opts = {}) {
      const overlayPath = opts.overlayPath || OVERLAY_PATH;
      return async function handle(req, res) {
        const body = req.body || {};
        if (body.node !== 'fc_controller') {
          return res.status(400).json({ error: `node ${JSON.stringify(body.node)} not allowlisted` });
        }
        let entries;
        if (Array.isArray(body.params)) entries = body.params;
        else if (body.param != null) entries = [{ param: body.param, value: body.value }];
        else return res.status(400).json({ error: 'body must include `param`+`value` or `params: []`' });
        if (entries.length === 0) return res.status(400).json({ error: 'params array is empty' });
        if (entries.length > 20) return res.status(400).json({ error: 'too many params (cap 20)' });

        for (const { param, value } of entries) {
          const v = cp.validate(param, value);
          if (!v.ok) return res.status(400).json({ error: v.reason, rejected_param: param });
        }

        try {
          const existing = await transport.read(overlayPath).catch(() => null);
          const parsed = existing ? yaml.load(existing) : {};
          const merged = mergeOverlay(parsed || {}, entries);
          const rendered = renderOverlay(merged);
          await transport.atomicWrite(overlayPath, rendered);
          return res.json({
            ok: true,
            persisted: entries.map(({ param, value }) => ({ param, value })),
            path: overlayPath,
          });
        } catch (e) {
          return res.status(500).json({ error: `persist failed: ${e.message}` });
        }
      };
    }

    module.exports = { OVERLAY_PATH, mergeOverlay, renderOverlay, makeHandler };
    ```

    function makeSshTransport({ host = '172.16.10.5', user = 'ubuntu', exec = require('child_process').execFile }) {
      const { promisify } = require('util');
      const pexec = promisify(exec);
      const sshArgs = (cmd) => ['-o', 'BatchMode=yes', '-o', 'StrictHostKeyChecking=accept-new', `${user}@${host}`, cmd];
      return {
        async read(p) {
          const { stdout } = await pexec('ssh', sshArgs(`cat ${p} 2>/dev/null || true`));
          return stdout || null;
        },
        async atomicWrite(p, content) {
          // Write tmp on bridge side, scp, then ssh-rename.
          const fs = require('fs').promises;
          const tmp = `/tmp/fc_overlay_${Date.now()}.yaml`;
          await fs.writeFile(tmp, content, 'utf8');
          try {
            await pexec('scp', ['-o', 'BatchMode=yes', '-o', 'StrictHostKeyChecking=accept-new', tmp, `${user}@${host}:${p}.tmp`]);
            await pexec('ssh', sshArgs(`mv -f ${p} ${p}.bak 2>/dev/null; mv -f ${p}.tmp ${p}`));
          } finally {
            await fs.unlink(tmp).catch(() => {});
          }
        },
      };
    }
    module.exports.makeSshTransport = makeSshTransport;

        async read(p) {

| bridge → fc1 (SSH or HTTP) | Layer 2 transport boundary; locked by spike.
