    ```javascript
    // Phase 28 D-17 Layer 1: live param tuning hot path.
    // Allowlist hardcoded here (Threat T-28-15: never load from disk).
    // ALL bridge SetParameters traffic flows through this module — index.js does not
    // touch rclnodejs SetParameters directly (Pitfall 8 — keep buffer-replay path clean).

    const NODE = 'fc_controller';
    const ALLOWED_NODES = new Set(['fc_controller']);

    // Param types per rcl_interfaces/msg/ParameterType (canonical ROS2 enum).
    const T_BOOL = 1, T_INTEGER = 2, T_DOUBLE = 3, T_STRING = 4;

    const DECLARED_MODES = ['fruiting', 'pinning'];  // hardcoded; new modes are deploys (D-03)
    const DEFEND_SIDES = new Set(['low', 'high', 'both']);

    function inRange(v, lo, hi) { return typeof v === 'number' && Number.isFinite(v) && v >= lo && v <= hi; }
    function inRangeOrNaN(v, lo, hi) { return (typeof v === 'number' && Number.isNaN(v)) || inRange(v, lo, hi); }

    // Allowlist: every entry is { type, validate(v) -> {ok, reason?} }
    function makeBandLowEntry()  { return { type: T_DOUBLE, validate: v => inRange(v, 0, 1) ? {ok:true} : {ok:false, reason:`band_low out of [0,1]`} }; }
    function makeBandHighEntry() { return { type: T_DOUBLE, validate: v => inRange(v, 0, 1) ? {ok:true} : {ok:false, reason:`band_high out of [0,1]`} }; }
    function makeTargetEntry()   { return { type: T_DOUBLE, validate: v => inRange(v, 0, 1) ? {ok:true} : {ok:false, reason:`target_humidity out of [0,1]`} }; }
    function makeDefendEntry()   { return { type: T_STRING, validate: v => DEFEND_SIDES.has(v) ? {ok:true} : {ok:false, reason:`defend_side must be low|high|both`} }; }
    function makeTtargetEntry()  { return { type: T_DOUBLE, validate: v => inRangeOrNaN(v, 0, 40) ? {ok:true} : {ok:false, reason:`t_target must be NaN or in [0,40]`} }; }

    const ALLOWLIST = (() => {
      const a = {};
      a['active_mode'] = { type: T_STRING, validate: v => DECLARED_MODES.includes(v) ? {ok:true} : {ok:false, reason:`active_mode ${JSON.stringify(v)} not in ${JSON.stringify(DECLARED_MODES)}`} };
      for (const m of DECLARED_MODES) {
        a[`modes.${m}.target_humidity`] = makeTargetEntry();
        a[`modes.${m}.band_low`] = makeBandLowEntry();
        a[`modes.${m}.band_high`] = makeBandHighEntry();
        a[`modes.${m}.defend_side`] = makeDefendEntry();
        a[`modes.${m}.t_target`] = makeTtargetEntry();
      }
      a['pid_kp'] = { type: T_DOUBLE, validate: v => inRange(v, 0, 5) ? {ok:true} : {ok:false, reason:`pid_kp out of [0,5]`} };
      a['pid_ki'] = { type: T_DOUBLE, validate: v => inRange(v, 0, 1) ? {ok:true} : {ok:false, reason:`pid_ki out of [0,1]`} };
      a['pid_kd'] = { type: T_DOUBLE, validate: v => inRange(v, 0, 20) ? {ok:true} : {ok:false, reason:`pid_kd out of [0,20]`} };
      return a;
    })();

    function validate(param, value) {
      const spec = ALLOWLIST[param];
      if (!spec) return { ok: false, reason: `param ${param} not allowlisted` };
      return spec.validate(value);
    }

    function toParamValue(param, value) {
      const spec = ALLOWLIST[param];
      if (!spec) throw new Error(`toParamValue called for non-allowlisted param ${param}`);
      switch (spec.type) {
        case T_BOOL:    return { type: T_BOOL,    bool_value: !!value };
        case T_INTEGER: return { type: T_INTEGER, integer_value: Math.trunc(value) };
        case T_DOUBLE:  return { type: T_DOUBLE,  double_value: Number(value) };
        case T_STRING:  return { type: T_STRING,  string_value: String(value) };
      }
      throw new Error(`unknown type ${spec.type} for ${param}`);
    }

    // handler factory: takes the (already-initialized) rclnodejs Node so tests can inject a mock.
    function makeHandler(rosNode, opts = {}) {
      const SET_PARAMS_TIMEOUT_MS = opts.timeoutMs || 3000;
      return async function handle(req, res) {
        const body = req.body || {};
        const node = body.node;
        if (!ALLOWED_NODES.has(node)) {
          return res.status(400).json({ error: `node ${JSON.stringify(node)} not allowlisted` });
        }

        // Normalize single + batched forms into one params array.
        let params;
        if (Array.isArray(body.params)) {
          params = body.params;
        } else if (body.param != null) {
          params = [{ param: body.param, value: body.value }];
        } else {
          return res.status(400).json({ error: 'body must include `param`+`value` or `params: []`' });
        }
        if (params.length === 0) {
          return res.status(400).json({ error: 'params array is empty' });
        }

        // Validate every entry BEFORE sending — atomicity at the bridge layer mirrors
        // rclpy's whole-batch atomicity (Pitfall 4).
        for (const { param, value } of params) {
          const v = validate(param, value);
          if (!v.ok) return res.status(400).json({ error: v.reason, rejected_param: param });
        }

        // Build the rcl SetParameters request as ONE call with all params (Pitfall 4).
        const reqMsg = {
          parameters: params.map(({ param, value }) => ({
            name: param,
            value: toParamValue(param, value),
          })),
        };

        let cli;
        try {
          cli = rosNode.createClient('rcl_interfaces/srv/SetParameters', `/${node}/set_parameters`);
        } catch (e) {
          return res.status(500).json({ error: `createClient failed: ${e.message}` });
        }

        try {
          const resp = await Promise.race([
            new Promise((resolve, reject) => {
              cli.sendRequest(reqMsg, (r) => r ? resolve(r) : reject(new Error('rclnodejs returned no response')));
            }),
            new Promise((_, reject) => setTimeout(() => reject(new Error(`SetParameters timeout after ${SET_PARAMS_TIMEOUT_MS}ms`)), SET_PARAMS_TIMEOUT_MS)),
          ]);
          // resp.results: [{successful, reason}, ...] in the same order as request.parameters.
          const failures = (resp.results || []).map((r, i) => ({ ok: r.successful, reason: r.reason, param: params[i].param }))
                                                .filter(x => !x.ok);
          if (failures.length > 0) {
            const f = failures[0];
            return res.status(422).json({ error: f.reason, rejected_param: f.param });
          }
          return res.json({ ok: true, applied: params.map(({ param, value }) => ({ param, value })) });
        } catch (e) {
          return res.status(500).json({ error: e.message });
        }
      };
    }

    module.exports = { ALLOWLIST, DECLARED_MODES, validate, toParamValue, makeHandler };
    ```
