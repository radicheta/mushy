Purpose: this is the seam where the farmer experiences the quote-threading mechanism. When commit-outcome lands or a confirm prompt fires, the bot's outgoing message Signal-quotes the farmer's original capture message so the ack renders attached to "that observation". Helper getCaptureQuoteTarget (confirm-db.js, new):
  async function getCaptureQuoteTarget(pool, captureId) {
    if (!captureId) return null;
    try {
      const r = await pool.query(
        'SELECT signal_msg_ts, sender, raw_text FROM signal_capture WHERE id = $1 LIMIT 1',
        [captureId]
      );
      const row = r.rows && r.rows[0];
      if (!row || row.signal_msg_ts == null) return null;
      return { signal_msg_ts: row.signal_msg_ts, sender: row.sender, raw_text: row.raw_text || '' };
    } catch (_e) {
      return null;
    }
  }
  Export from confirm-db.js. Never throws; null on any failure.

Helper tryBuildQuoteForDraft (outbound-confirm.js, file-local):
  async function tryBuildQuoteForDraft(pool, draftRow, logger) {
    if (!pool || !confirmDb || !draftRow) return null;
    const captureIds = Array.isArray(draftRow.source_capture_ids) ? draftRow.source_capture_ids : [];
    if (captureIds.length === 0) return null;
    const tgt = await confirmDb.getCaptureQuoteTarget(pool, captureIds[0]);
    if (!tgt) {
      logger.warn && logger.warn(`[outbound-confirm] no quote target for draft=${truncId(draftRow.id)} capture=${truncId(captureIds[0])} -- sending unquoted ack`);
      return null;
    }
    const trimmed = (tgt.raw_text || '').slice(0, 200);
    return { timestamp: Number(tgt.signal_msg_ts), author: tgt.sender, message: trimmed };
  }
  Local to outbound-confirm.js (not exported); consumes pool + confirmDb that the dispatcher already has in scope.
