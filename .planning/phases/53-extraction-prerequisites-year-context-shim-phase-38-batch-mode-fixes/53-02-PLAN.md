      const hasSeedingSession = draftsArr.some(d => d.draft && d.draft.type === 'seeding_session');

      if (inFlight) { await extractionDb.updateDraftStatus(pool, inFlight.id, DRAFT_STATUS.EXPIRED); }
      const results = [];

        const r = await processSingleDraft({

        results.push(r);
      }
