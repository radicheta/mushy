**3. Add pushFrame function (after camera state variables):**
```javascript
function pushFrame(jpegBuffer) {
    latestFrame = jpegBuffer;
    lastFrameTime = Date.now();
    const header = [
        `--${BOUNDARY}`,
        'Content-Type: image/jpeg',
        `Content-Length: ${jpegBuffer.length}`,
        '',
        ''
    ].join('\r\n');

    mjpegClients.forEach(res => {
        if (!res.writable) {
            mjpegClients.delete(res);
            return;
        }
        try {
            res.write(header, 'ascii');
            res.write(jpegBuffer);
            res.write('\r\n', 'ascii');
        } catch (e) {
            mjpegClients.delete(res);
        }
    });
}
```

**4. Add saveSnapshot function (after pushFrame):**
```javascript
function saveSnapshot() {
    if (!latestFrame) return;
    const now = new Date();
    const dateDir = now.toISOString().slice(0, 10); // YYYY-MM-DD
    const dir = path.join(SNAPSHOT_DIR, CAMERA_ID, dateDir);
    fs.mkdirSync(dir, { recursive: true });
    const filename = `${now.toISOString().replace(/[:.]/g, '-')}.jpg`;
    const filepath = path.join(dir, filename);
    fs.writeFile(filepath, latestFrame, (err) => {
        if (err) console.error('[camera] snapshot write failed:', err.message);
        else console.log(`[camera] snapshot saved: ${filepath} (${CAMERA_ID}, ${latestFrame.length} bytes)`);
    });
}
```

**5. Add MJPEG route (after the `/history/:topic` route, before the WebSocket section):**
```javascript
// Camera MJPEG stream endpoint (D-03)
app.get('/camera/mjpeg', (req, res) => {
    res.writeHead(200, {
        'Content-Type': `multipart/x-mixed-replace; boundary="${BOUNDARY}"`,
        'Cache-Control': 'no-cache, no-store',
        'Connection': 'close',
        'Pragma': 'no-cache'
    });
    mjpegClients.add(res);
    console.log(`[camera] MJPEG client connected (${mjpegClients.size} total)`);
    req.on('close', () => {
        mjpegClients.delete(res);
        console.log(`[camera] MJPEG client disconnected (${mjpegClients.size} total)`);
    });
});

// Camera latest frame endpoint (single JPEG for testing)
app.get('/camera/snapshot', (req, res) => {
    if (!latestFrame) {
        return res.status(503).json({ error: 'No camera frame available' });
    }
    res.writeHead(200, {
        'Content-Type': 'image/jpeg',
        'Content-Length': latestFrame.length,
        'Cache-Control': 'no-cache'
    });
    res.end(latestFrame);
});
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| ROS topic -> bridge | CompressedImage frames from VPN-only ROS network enter Node.js process |
| bridge -> HTTP client | MJPEG frames served to browser clients on LAN/VPN |
| bridge -> filesystem | Snapshot JPEG files written to host filesystem |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-04 | Denial of Service | MJPEG endpoint | mitigate | Track clients in Set, check `res.writable` before write, catch errors and remove stale clients. Log client count. Low risk: LAN/VPN only, few clients expected. |
| T-08-05 | Tampering | Snapshot path | accept | CAMERA_ID from env var (operator-controlled), not from HTTP request. `fs.mkdirSync` with `{ recursive: true }` uses safe path.join(). No user input in file path. |
| T-08-06 | Information Disclosure | /camera/mjpeg | accept | Endpoint accessible only via bridge port 8081 on WireGuard VPN. No authentication needed for mushroom camera on private network. Same trust level as existing /health and /history routes. |
| T-08-07 | Denial of Service | Snapshot disk | accept | 1 FPS * 30KB * 4/hr snapshots = ~120KB/hr disk usage. At 15-min intervals: ~3MB/day. Elder-plops has ample storage. No rotation needed in MVP. |
</threat_model>
