Run the following checks from the repo root on elder-plops and capture all outputs verbatim into `.planning/phases/16-system-health-panel/16-SMOKE-EVIDENCE.md`:

```bash
# 1. Which images are the containers running?
docker compose ps --format 'table {{.Name}}\t{{.Image}}\t{{.Status}}' | tee /tmp/16-smoke-ps.txt

# 2. /health has all new fields and no regressions
curl -s http://localhost:8081/health | python3 -m json.tool | tee /tmp/16-smoke-health.json

# 3. Served plugin.js has all six labels and ≥10 makeStatusLight hits
curl -s http://localhost:8080/plugins/fruiting-chamber/plugin.js > /tmp/16-served-plugin.js
grep -c makeStatusLight /tmp/16-served-plugin.js        # expect >= 10
for label in "'Sensors'" "'Camera feed'" "'Humidifier'" "'Bridge'" "'Pi reachable'" "'Grace'"; do
    if grep -q "$label" /tmp/16-served-plugin.js; then echo "LABEL_OK: $label"; else echo "LABEL_MISSING: $label"; fi
done

# 4. sensor_health WS broadcast actually flows (TRANSIENT_LOCAL should replay last message)
# If websocat is installed, capture one message; otherwise note "ws capture skipped".
if command -v websocat >/dev/null; then
    timeout 8 websocat ws://localhost:8081 2>/dev/null | grep -m1 'sensor_health' | tee /tmp/16-smoke-ws.txt || echo "no sensor_health message captured in 8s"
else
    echo "websocat not installed — WS smoke skipped, defer to human browser test"
fi

# 5. Bridge logs are clean
docker compose logs --tail=80 bridge | grep -E 'error|Error|ERROR|sensor_health|Sensor health' | tee /tmp/16-smoke-bridge-log.txt || true

# 6. openmct logs are clean
docker compose logs --tail=40 openmct | tee /tmp/16-smoke-openmct-log.txt || true
```
