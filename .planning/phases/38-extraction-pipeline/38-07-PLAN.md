    1. Create `test/eval/extraction/mushdatadump.test.js`. The driver:
       ```js
       'use strict';
       const path = require('path');
       const { createExtractor } = require('../../../src/extraction/extractor');
       const schemas = require('../../../src/extraction/schemas');
       const { loadFixtures } = require('./fixtures-loader');
       const scoring = require('./scoring');
       const { writeReport } = require('./report');

       const FIXTURE_DIR = process.env.EXTRACTION_FIXTURE_DIR || '/mnt/mossrock/shared/mushdatadump';
       const REPORT_PATH = path.resolve(__dirname, '../../../../../.planning/phases/38-extraction-pipeline/38-EVAL-REPORT.md');

       describe('mushdatadump eval (D-07 ship-gate)', () => {
         let fixtures;
         let extractor;
         beforeAll(() => {
           fixtures = loadFixtures(FIXTURE_DIR);
           extractor = createExtractor({
             apiKey: process.env.ANTHROPIC_API_KEY,
             logger: console,
             draftSchema: schemas.SubmissionSchema,
             draftJsonSchema: schemas.SUBMISSION_JSON_SCHEMA,
           });
         });

         test('runs all fixtures and writes report', async () => {
           const results = [];
           for (const fx of fixtures) {
             const r = await extractor.extract({
               captures: [{captureId: fx.imagePath, text: fx.expected.text || '', transcripts: fx.expected.transcript ? [fx.expected.transcript] : [], attachmentPaths: [fx.imagePath]}],
               inFlightDraft: null,
             });
             results.push({ fixture: fx, actual: r });
           }
           const scores = {
             schemaConformance: scoring.schemaConformance(results),
             requiredFieldMatch: scoring.exactFieldMatch(results),
             appropriateAskBack: scoring.appropriateAskBack(results),
             setEquality: scoring.setEquality(results),
             b5: scoring.b5PrecisionRecall(results),
             brier: scoring.brierScore(results),
             ece: scoring.ece(results),
             requiredFieldOrAppropriateAskBack: scoring.combinedFieldOrAskBack(results),
           };
           const verdict = (scores.schemaConformance >= 0.90 && scores.requiredFieldOrAppropriateAskBack >= 0.75) ? 'PASS' : 'FAIL';
           writeReport(REPORT_PATH, scores, fixtures.length, verdict);
           expect(scores.schemaConformance).toBeGreaterThanOrEqual(0);  // soft -- always passes the test framework
           // The HARD assertion is the report verdict line, not this jest test.
         }, 600000);
       });
       ```
