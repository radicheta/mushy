Equality assertion shape (Gray Area C):
  function equalsOnKeyFields(draft, gt) {
    if (draft.type !== gt.type) return false;
    if (draft.event_date !== gt.event_date) return false;
    if (draft.groups.length !== gt.groups.length) return false;
    for (let i=0; i<gt.groups.length; i++) {
      const g = draft.groups[i], t = gt.groups[i];
      if (g.parent.value !== t.parent.value) return false;
      if (g.species.value !== t.species.value) return false;
      if (g.qty.value !== t.qty.value) return false;
      const cd = g.child_block_names.value, ct = t.child_block_names.value;
      if (cd.length !== ct.length) return false;
      for (let j=0; j<ct.length; j++) if (cd[j] !== ct[j]) return false;
    }
    return true;
  }
