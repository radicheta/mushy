**1. Create `scripts/pi-deploy/test-dht22-raw.py`** — raw DHT22 test WITHOUT ROS (isolates sensor from stack):

```python
#!/usr/bin/env python3
"""Test DHT22 sensor directly on Pi GPIO — no ROS dependency.

Run on Pi: python3 test-dht22-raw.py
Tests GPIO pin 4 (BCM) — the DHT22 data pin from fc_config.yaml (dht_pin: 4).
"""
import sys
import time

# Try Adafruit DHT library (used by fc_sensors.py)
try:
    import adafruit_dht
    import board
except ImportError:
    print("ERROR: Adafruit DHT library not available.")
    print("Install with:")
    print("  sudo apt install -y libgpiod2")
    print("  pip3 install adafruit-circuitpython-dht")
    sys.exit(1)

DHT_PIN = board.D4  # GPIO4 (BCM) — matches fc_config.yaml dht_pin: 4

def main():
    print("=== DHT22 Raw Sensor Test (GPIO4) ===")
    print()

    sensor = adafruit_dht.DHT22(DHT_PIN)

    readings = []
    errors = 0
    max_attempts = 10

    print(f"Taking {max_attempts} readings (DHT22 can fail intermittently, this is normal)...")
    print()

    for i in range(max_attempts):
        try:
            temperature = sensor.temperature
            humidity = sensor.humidity

            if humidity is not None and temperature is not None:
                readings.append((temperature, humidity))
                print(f"  Reading {i+1}: temp={temperature:.1f}C  humidity={humidity:.1f}%")
            else:
                errors += 1
                print(f"  Reading {i+1}: None returned (sensor not ready)")

        except RuntimeError as e:
            errors += 1
            print(f"  Reading {i+1}: RuntimeError — {e} (normal for DHT22, retrying)")

        except Exception as e:
            errors += 1
            print(f"  Reading {i+1}: Unexpected error — {e}")

        time.sleep(2.5)  # DHT22 needs minimum 2 seconds between reads

    print()
    print("=== Results ===")
    print(f"Successful readings: {len(readings)}/{max_attempts}")
    print(f"Errors: {errors}/{max_attempts}")

    if len(readings) == 0:
        print()
        print("FAILED: No valid readings obtained.")
        print("Troubleshooting:")
        print("  1. Check DHT22 wiring: VCC=3.3V, GND=GND, DATA=GPIO4")
        print("  2. Check 10k pull-up resistor between DATA and VCC")
        print("  3. Try: sudo apt install -y libgpiod2")
        print("  4. Check permissions: user must be in 'gpio' group")
        sensor.exit()
        sys.exit(1)

    temps = [r[0] for r in readings]
    humids = [r[1] for r in readings]

    print(f"Temperature range: {min(temps):.1f}C — {max(temps):.1f}C")
    print(f"Humidity range: {min(humids):.1f}% — {max(humids):.1f}%")

    # Sanity checks
    sane = True
    for t, h in readings:
        if t < 0 or t > 60:
            print(f"WARNING: Temperature {t}C outside reasonable range (0-60)")
            sane = False
        if h < 0 or h > 100:
            print(f"WARNING: Humidity {h}% outside valid range (0-100)")
            sane = False

    if sane and len(readings) >= 3:
        print()
        print("=== DHT22 RAW TEST PASSED ===")
        print(f"Sensor is working. Got {len(readings)} valid readings.")
    elif sane:
        print()
        print("=== DHT22 RAW TEST MARGINAL ===")
        print(f"Only {len(readings)} valid readings — sensor may have intermittent issues.")
    else:
        print()
        print("=== DHT22 RAW TEST WARNING ===")
        print("Values outside expected range — check sensor and wiring.")

    sensor.exit()


if __name__ == "__main__":
    main()
```

**2. Create `scripts/pi-deploy/test-dht22-ros.sh`** — validates DHT22 through full ROS stack:

```bash
#!/usr/bin/env bash
set -euo pipefail

# Test DHT22 through ROS2 stack on Pi
# Prerequisites: fc-core service running (01-02 deploy workflow)

PI_HOST="${PI_HOST:-fc1}"
TIMEOUT=30
TOPIC="fc/humidity"

echo "=== DHT22 ROS2 Topic Validation ==="
echo "Host: ${PI_HOST}"
echo "Topic: ${TOPIC}"
echo "Timeout: ${TIMEOUT}s"
echo ""

# Step 1: Ensure fc_config.yaml has simulation_mode: false
echo "[1/4] Checking simulation_mode on Pi..."
SIM_MODE=$(ssh "${PI_HOST}" "grep 'simulation_mode' ~/mushroom_farm_ws/src/chambers/fc-core/config/fc_config.yaml" || echo "NOT_FOUND")
echo "  Config: ${SIM_MODE}"
if echo "${SIM_MODE}" | grep -q "true"; then
    echo ""
    echo "WARNING: simulation_mode is true. Switching to false for real hardware test..."
    ssh "${PI_HOST}" "sed -i 's/simulation_mode: true/simulation_mode: false/' ~/mushroom_farm_ws/src/chambers/fc-core/config/fc_config.yaml"
    echo "  Rebuilding and restarting..."
    ssh "${PI_HOST}" "cd ~/mushroom_farm_ws && source /opt/ros/jazzy/setup.bash && colcon build --packages-select fc_core --symlink-install"
    ssh "${PI_HOST}" "sudo systemctl restart fc-core"
    echo "  Waiting 10s for node startup..."
    sleep 10
fi

# Step 2: Check fc_sensors node is running
echo ""
echo "[2/4] Checking fc_sensors node..."
NODE_LIST=$(ssh "${PI_HOST}" "source /opt/ros/jazzy/setup.bash && source ~/mushroom_farm_ws/install/setup.bash && export ROS_DOMAIN_ID=69 && ros2 node list 2>/dev/null" || echo "NONE")
echo "  Active nodes: ${NODE_LIST}"
if ! echo "${NODE_LIST}" | grep -q "fc_sensors"; then
    echo "  ERROR: fc_sensors node not running. Check: ssh ${PI_HOST} 'journalctl -u fc-core -n 50'"
    exit 1
fi
echo "  fc_sensors node is running."

# Step 3: Check topic exists
echo ""
echo "[3/4] Checking ${TOPIC} topic..."
TOPICS=$(ssh "${PI_HOST}" "source /opt/ros/jazzy/setup.bash && source ~/mushroom_farm_ws/install/setup.bash && export ROS_DOMAIN_ID=69 && ros2 topic list 2>/dev/null" || echo "NONE")
if ! echo "${TOPICS}" | grep -q "${TOPIC}"; then
    echo "  ERROR: ${TOPIC} topic not found. Available topics:"
    echo "  ${TOPICS}"
    exit 1
fi
echo "  Topic ${TOPIC} exists."

# Step 4: Read actual humidity data
echo ""
echo "[4/4] Reading humidity data from ${TOPIC} (${TIMEOUT}s timeout)..."
echo "  (Capturing 5 messages...)"
DATA=$(ssh "${PI_HOST}" "source /opt/ros/jazzy/setup.bash && source ~/mushroom_farm_ws/install/setup.bash && export ROS_DOMAIN_ID=69 && timeout ${TIMEOUT} ros2 topic echo ${TOPIC} --once 2>/dev/null | head -20" || echo "TIMEOUT")

if [ "${DATA}" = "TIMEOUT" ] || [ -z "${DATA}" ]; then
    echo "  ERROR: No data received on ${TOPIC} within ${TIMEOUT}s"
    echo "  Check sensor wiring and logs: ssh ${PI_HOST} 'journalctl -u fc-core -n 50'"
    exit 1
fi

echo "  Received data:"
echo "  ${DATA}"
echo ""
echo "=== DHT22 ROS2 TOPIC TEST PASSED ==="
echo "Humidity data is being published to ${TOPIC} from real hardware."
```
