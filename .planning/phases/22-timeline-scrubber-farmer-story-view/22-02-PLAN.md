**burn_bar.js contract (concrete):**
```js
// src/mission-control/bridge/src/burn_bar.js
const { Jimp, JimpMime } = require('jimp');  // jimp v1 API; adjust import to `const Jimp = require('jimp');` if v1 default export differs — verify at install time
const { SANS_16_WHITE, measureText } = require('jimp');  // built-in bitmap font; fallback: use Jimp.loadFont(Jimp.FONT_SANS_16_WHITE) — confirm against installed jimp version

// Null/undefined -> '—' (en-dash, U+2014). Numbers -> toFixed(1).
function fmtNum(v) {
    if (v === null || v === undefined || Number.isNaN(v)) return '—';
    return Number(v).toFixed(1);
}

function fmtHum(v) {
    if (v === null || v === undefined) return '—';
    return v ? 'ON' : 'OFF';
}

// capturedAt: Date | ISO string. Returns the exact bar text string — pure, no side effects.
function formatBarText({ capturedAt, rh, temp, co2, hum }) {
    const iso = (capturedAt instanceof Date) ? capturedAt.toISOString() : String(capturedAt);
    return `${iso} · RH ${fmtNum(rh)}% · T ${fmtNum(temp)}°C · CO₂ ${fmtNum(co2)}ppm · HUM ${fmtHum(hum)}`;
}

// Bottom-bar layout constants (D-03 + planner's discretion block):
//   - Bar height: max(32px, 10% of image height)
//   - Background: black, opacity 0.55
//   - Text: white, jimp built-in SANS_16_WHITE (or SANS_32_WHITE if img height > 640)
//   - Horizontal padding: 8px
//   - Vertical centering inside the bar
async function burnBar(inputBuffer, barText) {
    const img = await Jimp.read(inputBuffer);
    const width = img.bitmap.width;
    const height = img.bitmap.height;
    const barH = Math.max(32, Math.round(height * 0.10));
    const barY = height - barH;

    // Semi-transparent black bar: create an overlay layer and composite.
    // jimp v1 supports new Jimp({width, height, color}) then opacity.
    const bar = new Jimp({ width, height: barH, color: 0x000000ff });
    bar.opacity(0.55);  // if jimp v1 uses .opacity differently, fall back to manually scanning the bar and multiplying alpha
    img.composite(bar, 0, barY);

    // Pick font size by height
    const fontKey = height >= 640 ? 'FONT_SANS_32_WHITE' : 'FONT_SANS_16_WHITE';
    const font = await Jimp.loadFont(Jimp[fontKey]);
    // Vertical center within the bar: (barH - lineHeight)/2 — use font.common.lineHeight if available; otherwise approximate with (barH/2 - 10).
    const textY = barY + Math.round((barH - (font.common?.lineHeight || 16)) / 2);
    img.print(font, 8, textY, barText, width - 16);

    return await img.getBuffer(JimpMime.jpeg, { quality: 85 });
}

module.exports = { formatBarText, burnBar };
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| ROS2 → bridge | `/fc1/camera/compressed` buffer becomes the raw input to jimp; malformed JPEG could crash the burn path |
| latestTelemetry → burnt bar | Sensor cache is trusted (internal ROS topics); no user input reaches the overlay text |
| retention.js FS unlink | Mirror-delete logic derives burnt path from raw `file_path` (stored in DB); DB is trusted but `file_path` is still a server-written value |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-22-05 | Denial-of-service | burn path blocking raw writes | mitigate | Burnt write is fire-and-forget inside an IIFE; errors logged only; raw write + DB insert never await the burn. `rawBuf = latestFrame` pin prevents aliasing. |
| T-22-06 | Tampering | overlay text injection | mitigate | `formatBarText` only receives values from internal `latestTelemetry` cache (ROS topics, not HTTP). `fmtNum` coerces via `Number(v).toFixed(1)` — non-numeric → `'—'`. No user-controlled data in the bar. |
| T-22-07 | Denial-of-service | jimp OOM on malformed JPEG | accept | Input is always `latestFrame` from the bridge's own ROS subscription. If fc_camera publishes corrupted data, jimp may throw — caught by the try/catch inside the IIFE, logged, raw+DB path unaffected. |
| T-22-08 | Tampering | burnt-path computation in retention | mitigate | `rawPath.startsWith(rawDir)` guard before swap prevents accidental unlink outside `SNAPSHOT_BURNT_DIR`. Plan 22-01 startup guard rejects `SNAPSHOT_BURNT_DIR == SNAPSHOT_DIR`, so swap cannot collapse. |
| T-22-09 | Information disclosure | stale rawBuf pin leaking across frames | accept | `rawBuf = latestFrame` captures the pointer to the exact buffer written as raw. No cross-frame leak: each saveSnapshot invocation gets its own pin. |
| T-22-10 | Denial-of-service | burnt tree fills disk faster than raw | mitigate | Same 5-min cadence as raw; ~55 MB/yr overhead (CONTEXT L79). Retention prune mirrors deletes in lockstep. If disk fills anyway, raw writes remain the source of truth — `/camera/frame` default mode returns 404 (gap over noise) and farmOS can render "no frame in this window". |
</threat_model>
