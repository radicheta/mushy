      function renderSeedingSession(draft) {
        if (draft.needs_input === 'starting_seq') return askBackBranch(draft);
        const groups = draft.groups || [];
        const totalChildren = groups.reduce((s,g)=> s + (g.child_block_names?.value?.length ?? g.qty?.value ?? 0), 0);
        const header = `Inoc session: ${draft.event_date}`;
        const summary = `${fmtNum(totalChildren)} blocks across ${fmtNum(groups.length)} parents`;
        const visible = groups.slice(0, 5);
        const overflow = groups.length > 5 ? `... (${fmtNum(groups.length - 5)} more groups)` : null;
        const rows = visible.map((g, i) => formatRow(i+1, g));
        // Compute column widths from rows + column-header row
        const widths = computeWidths([['KEY','PARENT','SPECIES','QTY','CHILDREN'], ...rows]);
        const colHeader = padRow(['KEY','PARENT','SPECIES','QTY','CHILDREN'], widths);
        const tableLines = rows.map(r => padRow(r, widths));
        const lines = [header, summary, '', colHeader, ...tableLines];
        if (overflow) lines.push(overflow);
        if (draft.notes) lines.push('', `note: ${draft.notes}`);
        lines.push('', 'YES to commit | NO to cancel | EDIT to change');
        return sanitizeFarmerText(lines.join('\n'));
      }

      function formatRow(key, g) {
        return [
          String(key),
          String(g.parent.value),
          String(g.species.value),
          fmtNum(g.qty.value),
          renderChildren(g.child_block_names.value, g.species.value),
        ];
      }

      function renderChildren(names, strain) {
        if (!Array.isArray(names) || names.length === 0) return '';
        if (names.length <= 2) return names.join(', ');
        // Extract SEQs assuming BLOCK_NAME_RE NNNNNN_XXX_SEQ
        const seqs = names.map(n => parseSeq(n)); // returns int or null
        if (seqs.every(s => s !== null)) {
          const sorted = [...seqs].sort((a,b)=>a-b);
          const isConsecutive = sorted.every((s,i)=> i===0 || s === sorted[i-1]+1);
          if (isConsecutive) {
            const prefix = names[0].slice(0, names[0].lastIndexOf('_')+1); // '260522_KOY_'
            return `${prefix}${sorted[0]}..${sorted[sorted.length-1]}`;
          }
        }
        return names.join(', ');
      }

      function parseSeq(name) {
        const m = /_(\d+)$/.exec(name);
        return m ? parseInt(m[1], 10) : null;
      }

      function computeWidths(rows) { /* per-column max length, with min(KEY=4, PARENT=15, SPECIES=8, QTY=4) */ }
      function padRow(cells, widths) { return cells.map((c,i) => i === cells.length-1 ? c : c.padEnd(widths[i] + 2, ' ')).join(''); }
