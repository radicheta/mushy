PRE-PROCESSING CHAIN MIRROR (Pitfall 3):
Production fc_controller.humidity_callback (line 180-186) does:

In open-loop replay, no PID, so no preprocessing — compare raw model RH

```python
"""Offline replay loop for fc_core sim — TWO STRICTLY SEPARATE MODES.

Open-loop  : feed historical duty(t) + T_amb(t); observe model RH(t).
             Validates the PLANT model.
Closed-loop: feed only T_amb(t) + target(t); let compute_duty() drive duty
             from modeled RH; observe model RH(t).
             Validates the CONTROLLER against the modeled plant.

DO NOT mix them. Different fixtures, different acceptance criteria.
See RESEARCH.md "Pitfall 5: Replaying historical actuator commands AND
closing the loop is a category error".
"""

import argparse, csv, math, random
from collections import deque
from statistics import median
from fc_core.control_kernel import compute_duty
from fc_core.vendor.simple_pid import PID
from fc_core.sim.chamber_model import ChamberState, ChamberParams, step
from fc_core.sim.pwm_window import PwmWindow


def _read_csv(path):
    with open(path) as f:
        return list(csv.DictReader(f))


def run_open_loop(csv_path, params=None):
    """Feed historical duty(t) + T_amb(t) to the model. PID is NOT used."""
    params = params or ChamberParams()
    rows = _read_csv(csv_path)
    state = ChamberState(rh=float(rows[0]['rh_measured']), m_water_g=70.0)
    out = []
    prev_t = float(rows[0]['t_s'])
    for r in rows:
        t = float(r['t_s'])
        dt = t - prev_t
        if dt <= 0:
            continue
        state = step(state, params,
                     duty=float(r['duty_measured']),
                     t_amb_c=float(r['t_amb_c']),
                     dt_s=dt)
        out.append((t, state.rh,
                    float(r['rh_measured']),
                    float(r['t_amb_c']),
                    float(r['duty_measured'])))
        prev_t = t
    return out


def run_closed_loop(csv_path, gains, bypass_threshold=0.025,
                    params=None, sensor_noise_std=0.0004,
                    median_filter_n=5, seed=42):
    """Feed only T_amb(t) + target(t); compute_duty drives the chamber."""
    params = params or ChamberParams()
    rng = random.Random(seed)
    rows = _read_csv(csv_path)
    state = ChamberState(rh=float(rows[0]['rh_measured']), m_water_g=70.0)
    pid = PID(*gains, setpoint=0.0, sample_time=None,
              output_limits=(0.0, 1.0), auto_mode=False,
              proportional_on_measurement=False,
              differential_on_measurement=True)
    pid.set_auto_mode(True, last_output=0.15)  # mirror _engage_pid_bumplessly
    pwm = PwmWindow(window_s=120.0, min_pulse_s=10.0, max_avg=0.90)
    rh_buffer = deque(maxlen=median_filter_n)
    out = []
    prev_t = float(rows[0]['t_s'])
    for r in rows:
        t = float(r['t_s'])
        dt = t - prev_t
        if dt <= 0:
            continue
        target = float(r['target'])
        t_amb = float(r['t_amb_c'])

        # Mirror prod preprocessing: noise + median-of-5
        rh_observed = state.rh + rng.gauss(0.0, sensor_noise_std)
        rh_buffer.append(rh_observed)
        rh_filtered = median(rh_buffer)

        decision = compute_duty(pid=pid,
                                current_humidity=rh_filtered,
                                effective_setpoint=target,
                                bypass_threshold=bypass_threshold,
                                dt=dt)
        # Pump duty through PwmWindow → relay state → effective duty for the model.
        # The model's own dead-time queue handles θ.
        relay = pwm.tick(decision.duty, dt)
        effective_duty = 1.0 if relay else 0.0
        state = step(state, params,
                     duty=effective_duty,
                     t_amb_c=t_amb, dt_s=dt)
        p, i, d = pid.components
        out.append((t, state.rh, target, decision.duty, t_amb, p, i, d))
        prev_t = t
    return out


def main(argv=None):
    ap = argparse.ArgumentParser()
    ap.add_argument('--mode', choices=['open', 'closed'], required=True)
    ap.add_argument('--csv', required=True)
    ap.add_argument('--gains', help='Kp,Ki,Kd (closed-loop only)')
    ap.add_argument('--output', help='write rows to this CSV; if omitted, stdout')
    args = ap.parse_args(argv)
    if args.mode == 'open':
        rows = run_open_loop(args.csv)
    else:
        if not args.gains:
            ap.error('--gains required for closed-loop')
        kp, ki, kd = map(float, args.gains.split(','))
        rows = run_closed_loop(args.csv, gains=(kp, ki, kd))
    sink = open(args.output, 'w') if args.output else None
    writer = csv.writer(sink or __import__('sys').stdout)
    writer.writerow(['t_s', 'rh_modeled', '...rest...'])
    for row in rows:
        writer.writerow(row)
    if sink:
        sink.close()
    return 0


if __name__ == '__main__':
    raise SystemExit(main())
```
