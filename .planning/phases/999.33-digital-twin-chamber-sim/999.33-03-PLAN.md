    2. Sample ambient T from synthetic_uruguay_autumn(sim_t_s) where

```python
"""24-hour synthetic grow — CI gate for closed-loop sim fidelity.

Reuses run_closed_loop's logic but drives with synthetic disturbances
instead of a CSV. Pure Python, zero ROS imports.
"""

import math, random
from collections import deque
from statistics import median
from fc_core.control_kernel import compute_duty
from fc_core.vendor.simple_pid import PID
from fc_core.sim.chamber_model import ChamberState, ChamberParams, step
from fc_core.sim.pwm_window import PwmWindow
from fc_core.sim.disturbance import synthetic_uruguay_autumn


def run(gains, target=0.94, duration_s=86400, dt=1.0,
        bypass_threshold=0.025, sensor_noise_std=0.0004,
        params=None, seed=42, initial_rh=0.94):
    """Closed-loop 24h grow against synthetic Uruguay-autumn ambient.

    Returns: list of (t_s, rh, target, duty, t_amb).
    """
    params = params or ChamberParams()
    rng = random.Random(seed)
    state = ChamberState(rh=initial_rh, m_water_g=70.0)
    pid = PID(*gains, setpoint=0.0, sample_time=None,
              output_limits=(0.0, 1.0), auto_mode=False,
              proportional_on_measurement=False,
              differential_on_measurement=True)
    pid.set_auto_mode(True, last_output=0.15)
    pwm = PwmWindow(window_s=120.0, min_pulse_s=10.0, max_avg=0.90)
    rh_buffer = deque(maxlen=5)
    rows = []
    n_ticks = int(duration_s / dt)
    for tick in range(n_ticks):
        t = tick * dt
        t_amb = synthetic_uruguay_autumn(t)
        rh_observed = state.rh + rng.gauss(0.0, sensor_noise_std)
        rh_buffer.append(rh_observed)
        rh_filtered = median(rh_buffer) if rh_buffer else rh_observed
        decision = compute_duty(pid=pid,
                                current_humidity=rh_filtered,
                                effective_setpoint=target,
                                bypass_threshold=bypass_threshold,
                                dt=dt)
        relay = pwm.tick(decision.duty, dt)
        effective_duty = 1.0 if relay else 0.0
        state = step(state, params,
                     duty=effective_duty, t_amb_c=t_amb, dt_s=dt)
        rows.append((t, state.rh, target, decision.duty, t_amb))
    return rows
```
