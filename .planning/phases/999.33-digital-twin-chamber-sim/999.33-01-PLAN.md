```python
from dataclasses import dataclass
from fc_core.vendor.simple_pid import PID

@dataclass(frozen=True)
class ControlDecision:
    duty: float            # final duty written to humidifier_duty topic
    raw_pid_output: float  # pre-clamp, for telemetry
    bypass_active: bool    # True if Mode C took the duty=1.0 branch

def compute_duty(
    pid: PID,
    current_humidity: float,      # fraction (0.94 = 94 %)
    effective_setpoint: float,    # fraction
    bypass_threshold: float,      # fraction (e.g. 0.025 = 2.5 %)
    dt: float,                    # seconds; MUST be explicit (Pitfall 1)
) -> ControlDecision:
    error_pct = (current_humidity - effective_setpoint) * 100.0
    bypass_pct = bypass_threshold * 100.0
    if abs(error_pct) > bypass_pct:
        if pid.auto_mode:
            pid.set_auto_mode(False)
        return ControlDecision(duty=1.0, raw_pid_output=1.0, bypass_active=True)
    if not pid.auto_mode:
        pid.set_auto_mode(True, last_output=1.0)
    raw = pid(error_pct, dt=dt)
    return ControlDecision(duty=raw, raw_pid_output=raw, bypass_active=False)
```
